forEach() – Display each user
------------------------------------------------------------------------------------
    users.forEach(user => {

map() – Get only user names
----------------------------------------------------------------------------------
const names = users.map(user => user.name);

filter() – Find users above age 22
------------------------------------------------------------------------------------------------
const result = users.filter(user => user.age > 22);

find() – Find a particular user
----------------------------------------------------------------------------------
const user = users.find(user => user.id === 2);

querySelectorAll()	                Selects all matching elements

remove()	                        Removes an element
setAttribute()	                    Sets an HTML attribute
getAttribute()	                    Gets an HTML attribute
classList.add()	                    Adds a CSS class
